\begin{problem}{N.1}{Example problem}{0}
  Let $X$ and $Y$ be discrete random variables. Show that
  \[ \P(X \given Y) = \frac{\P(Y \given X)\,\P(X)}{\P(Y)}. \]
  \begin{enumerate}
    \item First sub-part.
    \item Second sub-part.
  \end{enumerate}
\end{problem}

\begin{solution}
  \part{a}
  By the product rule,
  \begin{align*}
    \P(X, Y) &= \P(X \given Y)\,\P(Y) = \P(Y \given X)\,\P(X).
  \end{align*}

  \part{b}
  Example belief network:
  \begin{center}
    \begin{tikzpicture}
      \node[bnnode] (X) {$X$};
      \node[bnnode, below left=1cm and 0.6cm of X] (Y) {$Y$};
      \node[bnnode, below right=1cm and 0.6cm of X] (Z) {$Z$};
      \draw[bnedge] (X) -- (Y);
      \draw[bnedge] (X) -- (Z);
    \end{tikzpicture}
  \end{center}
\end{solution}
